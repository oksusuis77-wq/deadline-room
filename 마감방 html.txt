        // --- Firebase Init ---
        async function initApp() {
            try {
                try { app = initializeApp(firebaseConfig); } catch(e) {}
                auth = getAuth(app);
                db = getFirestore(app);

                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                    const cred = await signInWithCustomToken(auth, __initial_auth_token);
                    currentUser = cred.user;
                } else {
                    const cred = await signInAnonymously(auth);
                    currentUser = cred.user;

                document.getElementById('auth-status').classList.replace('bg-yellow-400', 'bg-emerald-500');
                document.getElementById('user-info').textContent = `${myNickname} (${currentUser.uid.substring(0,4)})`;
                
                document.getElementById('log-date').valueAsDate = new Date();

                setupLogsListener();

                console.error("Auth Error:", error);

                showToast("연결 중 오류가 발생했습니다.");
            }
        }

        async function initUserProfile() {
            if (!currentUser) return;
            const userRef = doc(db, 'artifacts', appId, 'public', 'data', 'users', currentUser.uid);
            const snap = await getDoc(userRef);
            
            if (!snap.exists()) {
                await setDoc(userRef, {
                    nickname: myNickname,
                    goal: '',
                    dailySeconds: 0,
                    totalChars: 0,
                    warnings: 0,
                    streak: 0,
                    weeklyDays: 0,
                    monthlyHours: 0,
                    monthlyChars: 0,
                    lastActive: serverTimestamp(),
                    isOnline: true
                });
                dailySharedSeconds = 0;
            } else {
                await updateDoc(userRef, {
                    nickname: myNickname,
                    isOnline: true,
                    lastActive: serverTimestamp()
                });
                
                if (snap.data().goal) document.getElementById('room-goal-input').value = snap.data().goal;
                dailySharedSeconds = snap.data().dailySeconds || 0;
            }
            updateProgressUI();
        }

                        renderCommunityLog();
                    }
                },
                error: (e) => console.error(e)
            });
        }

        async function startStream(type) {
            if (mediaStream) stopStream();
            try {
                if (type === 'cam') {
                    mediaStream = await navigator.mediaDevices.getUserMedia({ video: true, audio: false });
                } else {
                    mediaStream = await navigator.mediaDevices.getDisplayMedia({ video: true, audio: false });
                }

                videoElement.srcObject = mediaStream;
                videoElement.classList.remove('hidden-safely');
                videoPlaceholder.classList.add('hidden-safely');
                
                document.getElementById('btn-cam').classList.add('hidden-safely');
                document.getElementById('btn-screen').classList.add('hidden-safely');
                document.getElementById('btn-stop').classList.remove('hidden-safely');
                liveIndicator.classList.remove('hidden-safely');

                mediaStream.getVideoTracks()[0].onended = stopStream;
                startTimeTracking();
                showToast(type === 'cam' ? "웹캠 공유 시작" : "화면 공유 시작");

                showToast("권한이 거부되었거나 장치가 없습니다.");
            }
        }

        // Time Tracking & Sync
        function updateProgressUI() {
            const h = Math.floor(dailySharedSeconds / 3600).toString().padStart(2, '0');
            const m = Math.floor((dailySharedSeconds % 3600) / 60).toString().padStart(2, '0');
            const s = (dailySharedSeconds % 60).toString().padStart(2, '0');
            
            document.querySelectorAll('.daily-time-display').forEach(el => el.textContent = `${h}:${m}:${s}`);
            
            const percent = Math.min((dailySharedSeconds / GOAL_SECONDS) * 100, 100);
            document.getElementById('progress-bar').style.width = `${percent}%`;
            document.getElementById('progress-percent').textContent = `${Math.floor(percent)}%`;
        }

        function startTimeTracking() {
            if (sharingInterval) return;
            sharingInterval = setInterval(() => {
                dailySharedSeconds++;
                updateProgressUI();
                if (dailySharedSeconds % 10 === 0) saveDailyProgress(); // Sync every 10 seconds for real-time feel
            }, 1000);
        }

        function stopTimeTracking() {
            if (sharingInterval) {
                clearInterval(sharingInterval);
                sharingInterval = null;
                saveDailyProgress();
            }
        }

        async function saveDailyProgress() {
            if (!currentUser) return;
            const goalInput = document.getElementById('room-goal-input').value;
            const userRef = doc(db, 'artifacts', appId, 'public', 'data', 'users', currentUser.uid);
            await updateDoc(userRef, { 
                dailySeconds: dailySharedSeconds, 
                goal: goalInput,
                lastActive: serverTimestamp()
            }).catch(e => console.error("Sync Error:", e));
        }

        document.getElementById('btn-save-goal').addEventListener('click', () => {
            saveDailyProgress();
            showToast("오늘의 목표가 저장되었습니다.");
        });

        // ==================== POMODORO LOGIC ====================
        const POMO_WORK = 25 * 60;
        const POMO_BREAK = 5 * 60;
        let pomoSeconds = POMO_WORK;
        let isPomoRunning = false;
        let pomoInterval = null;
        let pomoMode = 'work';

        const pomoDisplay = document.getElementById('pomo-display');
        const btnPomoStart = document.getElementById('btn-pomo-start');
        const pomoBgFill = document.getElementById('pomo-bg-fill');
        const mWork = document.getElementById('mode-work');
        const mBreak = document.getElementById('mode-break');

        function updatePomoUI() {
            const m = Math.floor(pomoSeconds / 60).toString().padStart(2, '0');
            const s = (pomoSeconds % 60).toString().padStart(2, '0');
            pomoDisplay.textContent = `${m}:${s}`;
            
            const total = pomoMode === 'work' ? POMO_WORK : POMO_BREAK;
            const progress = ((total - pomoSeconds) / total) * 100;
            pomoBgFill.style.transform = `scaleY(${progress / 100})`;
            
            if(pomoMode === 'work') {
                pomoBgFill.classList.replace('bg-green-50', 'bg-gray-50');
            } else {
                pomoBgFill.classList.replace('bg-gray-50', 'bg-green-50');
            }
        }

        function setPomoMode(mode) {
            if(isPomoRunning) togglePomo();
            pomoMode = mode;
            pomoSeconds = mode === 'work' ? POMO_WORK : POMO_BREAK;
            
            if (mode === 'work') {
                mWork.className = 'px-4 py-1.5 rounded-full text-sm font-bold bg-gray-900 text-white transition-all shadow-md';
                mBreak.className = 'px-4 py-1.5 rounded-full text-sm font-bold bg-transparent text-gray-500 hover:bg-gray-100 transition-all';
            } else {
                mBreak.className = 'px-4 py-1.5 rounded-full text-sm font-bold bg-gray-900 text-white transition-all shadow-md';
                mWork.className = 'px-4 py-1.5 rounded-full text-sm font-bold bg-transparent text-gray-500 hover:bg-gray-100 transition-all';
            }
            updatePomoUI();
        }

        function togglePomo() {
            if (isPomoRunning) {
                clearInterval(pomoInterval);
                isPomoRunning = false;
                btnPomoStart.textContent = "계속하기";
            } else {
                isPomoRunning = true;
                btnPomoStart.textContent = "일시정지";
                pomoInterval = setInterval(() => {
                    if (pomoSeconds > 0) {
                        pomoSeconds--;
                        updatePomoUI();
                    } else {
                        showToast(pomoMode === 'work' ? "휴식 시간!" : "집중 시간!");
                        setPomoMode(pomoMode === 'work' ? 'break' : 'work');
                    }
                }, 1000);
            }
        }

        mWork.addEventListener('click', () => setPomoMode('work'));
        mBreak.addEventListener('click', () => setPomoMode('break'));
        btnPomoStart.addEventListener('click', togglePomo);
        document.getElementById('btn-pomo-reset').addEventListener('click', () => setPomoMode(pomoMode));

        function setupLogsListener() {
            if (!currentUser) return;
            const logsRef = collection(db, 'artifacts', appId, 'users', currentUser.uid, 'logs');
            onSnapshot(query(logsRef), {
                next: (snap) => {
                    const logs = [];
                    snap.forEach(d => logs.push({ id: d.id, ...d.data() }));
                    logs.sort((a, b) => (b.timestamp || 0) - (a.timestamp || 0));
                    
                    const todayStr = new Date().toDateString();
                    myTotalCharsToday = logs
                        .filter(l => (new Date(l.date || l.timestamp).toDateString() === todayStr))
                        .reduce((sum, l) => sum + (parseInt(l.count) || 0), 0);
                        
                    renderLogs(logs);
                },
                error: (e) => console.error("Logs error:", e)
            });
        }

        function renderLogs(logs) {
            const list = document.getElementById('log-list');
            list.innerHTML = '';
            if (!logs.length) {
                list.innerHTML = '<div class="break-inside-avoid text-center text-gray-400 py-10 text-sm w-full">아직 작성된 일지가 없습니다.</div>';
                return;
            }

            logs.forEach(log => {
                const dateStr = log.date ? log.date : new Date(log.timestamp).toLocaleDateString('ko-KR');
                const el = document.createElement('div');
                el.className = 'bg-white rounded-3xl p-6 card-shadow border border-gray-100 break-inside-avoid relative group mb-6 inline-block w-full';
                el.innerHTML = `
                    <div class="flex justify-between items-start mb-2">
                        <h3 class="text-lg font-bold text-gray-900">${escapeHTML(log.title || '제목 없음')}</h3>
                        <span class="text-xs font-medium text-gray-400 bg-gray-50 px-2 py-1 rounded-md font-[Pretendard]">${dateStr}</span>
                    </div>
                    <div class="mb-4">
                        <span class="inline-block bg-gray-100 text-gray-800 text-xs font-bold px-3 py-1.5 rounded-xl border border-gray-200">${log.count}자 작성</span>
                    </div>
                    <p class="text-gray-700 leading-relaxed whitespace-pre-wrap">${escapeHTML(log.note)}</p>
                    <button class="del-log-btn absolute top-4 right-4 opacity-0 group-hover:opacity-100 text-red-300 hover:text-red-500 transition-opacity bg-white rounded-full p-1 shadow-sm" data-id="${log.id}">
                        <svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M6 18L18 6M6 6l12 12"></path></svg>
                    </button>
                `;
                list.appendChild(el);
            });

            list.querySelectorAll('.del-log-btn').forEach(btn => {
                btn.addEventListener('click', async (e) => {
                    if(!currentUser) return;
                    const id = e.currentTarget.dataset.id;
                    await deleteDoc(doc(db, 'artifacts', appId, 'users', currentUser.uid, 'logs', id));
                    showToast("기록이 삭제되었습니다.");
                });
            });
        }

        document.getElementById('log-form').addEventListener('submit', async (e) => {
            e.preventDefault();
            if (!currentUser) return;
            const date = document.getElementById('log-date').value;
            const title = document.getElementById('log-title').value;
            const count = parseInt(document.getElementById('log-count').value);
            const note = document.getElementById('log-note').value;
            
            try {
                // Save Private Log
                await addDoc(collection(db, 'artifacts', appId, 'users', currentUser.uid, 'logs'), { 
                    date, title, count, note, timestamp: Date.now() 
                });
                
                // Update Public Stats
                const userRef = doc(db, 'artifacts', appId, 'public', 'data', 'users', currentUser.uid);
                const snap = await getDoc(userRef);
                if (snap.exists()) {
                    await updateDoc(userRef, {
                        totalChars: (snap.data().totalChars || 0) + count,
                        monthlyChars: (snap.data().monthlyChars || 0) + count,
                        lastActive: serverTimestamp()
                    });
                }

                e.target.reset();
                document.getElementById('log-date').valueAsDate = new Date();
                showToast("일지가 저장되었습니다.");
            } catch (err) { console.error(err); }
        });

        }
